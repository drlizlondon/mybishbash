## Step 1b — paste the "CHECK" tool (you'll reuse it all test)

After the reload, paste this **once**. It does nothing visible — it just teaches
the Console a command called `CHECK()` that prints all the storage values in a
neat table so you never have to dig through menus.

```js
// --- paste once: teaches CHECK() ---
async function CHECK(){
  const idb = await new Promise((res)=>{ const r=indexedDB.open("mybishbash"); r.onsuccess=()=>{ const db=r.result; if(!db.objectStoreNames.contains("kv")){db.close();return res({note:"not migrated yet"});} const o={}; const tx=db.transaction(["kv","meta"],"readonly"); const c=tx.objectStore("kv").get("mybishbash.cards.v1"); c.onsuccess=()=>{try{o.cardTitle=JSON.parse(c.result)[0]?.promptText;}catch{o.cardTitle=c.result;}}; const m=tx.objectStore("meta").get("migratedFromLocalStorage"); m.onsuccess=()=>{o.marker=m.result?"PRESENT":"missing";}; tx.oncomplete=()=>{db.close();res(o);}; tx.onerror=()=>{db.close();res({note:"read error"});}; }; r.onerror=()=>res({note:"no database"}); });
  let ls="(none)"; try{ ls=JSON.parse(localStorage.getItem("mybishbash.cards.v1"))[0]?.promptText; }catch{}
  const t={ "IndexedDB card": idb.cardTitle??idb.note, "localStorage card": ls, "Migrated marker": idb.marker??idb.note, "Engine switch": localStorage.getItem("mybishbash.storage-engine.v1")||"(default IndexedDB)", "Sync retry REQUEST": localStorage.getItem("mybishbash.storage-migration-retry.v1")||"(none)", "Sync retry ACK": localStorage.getItem("mybishbash.storage-migration-retry-ack.v1")||"(none)" };
  console.table(t); return t;
}
console.log("CHECK() ready — type CHECK() any time");
```
